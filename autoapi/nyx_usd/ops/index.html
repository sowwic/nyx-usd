    ops

    ops

    USD operations run by Nyx nodes in an interpreter that has pxr.

    Plain functions that take and return JSON data. Nothing here imports Nyx or the bridge, so the package works in any application shipping pxr (an OpenUSD build, Maya, Houdini, Blender).

    A stage is identified by the identifier of its root layer, a string. The opened stages are kept in this package until they are closed, so one node can edit what the previous one opened. They stay open until close_stage or close_all, or the process ends.

    The nodes call nyx_usd.ops.<name>, so every operation is imported here.

    def copy_prim(stage: str, prim_path: str, new_path: str) -> str:
    """Copy a prim with everything below it, in the layer the stage edits.

    Args:
        stage (str): stage handle.
        prim_path (str): path of the prim.
        new_path (str): path of the copy. The prims above it are created if missing.

    Raises:
        ValueError: no prim at the path, or the layer has no spec to copy.

    Returns:
        str: handle of the stage.
    """
    get_prim(stage, prim_path)
    source, target = Sdf.Path(prim_path), Sdf.Path(new_path)
    layer = get_stage(stage).GetEditTarget().GetLayer()
    if not layer.GetPrimAtPath(source):
        raise ValueError(f"{prim_path} has no spec in {layer.identifier}")
    Sdf.CreatePrimInLayer(layer, target.GetParentPath())
    if not Sdf.CopySpec(layer, source, layer, target):
        raise ValueError(f"Can't copy {prim_path} to {new_path}")
    return stage

    def create_material(
    stage: str, prim_path: str, diffuse_color: list[float] | None = None
) -> str:
    """Define a material with a UsdPreviewSurface shader.

    Args:
        stage (str): stage handle.
        prim_path (str): path of the material. The shader is `PreviewSurface` below it.
        diffuse_color (list[float] | None): red, green and blue, `[0.8, 0.8, 0.8]` if
            omitted.

    Raises:
        ValueError: the color doesn't have three numbers.

    Returns:
        str: handle of the stage.
    """
    color = _vector([0.8, 0.8, 0.8] if diffuse_color is None else diffuse_color, "color")
    usd_stage = get_stage(stage)
    material = UsdShade.Material.Define(usd_stage, prim_path)
    shader = UsdShade.Shader.Define(usd_stage, f"{prim_path}/PreviewSurface")
    shader.CreateIdAttr("UsdPreviewSurface")
    shader.CreateInput("diffuseColor", Sdf.ValueTypeNames.Color3f).Set(Gf.Vec3f(*color))
    surface = shader.CreateOutput("surface", Sdf.ValueTypeNames.Token)
    material.CreateSurfaceOutput().ConnectToSource(surface)
    return stage

    def create_mesh(
    stage: str,
    prim_path: str,
    points: list[list[float]],
    face_vertex_counts: list[int],
    face_vertex_indices: list[int],
) -> str:
    """Define a polygon mesh prim.

    Args:
        stage (str): stage handle.
        prim_path (str): path of the prim.
        points (list[list[float]]): position of every vertex, as x, y and z.
        face_vertex_counts (list[int]): number of vertices of every face.
        face_vertex_indices (list[int]): vertex of every corner of every face.

    Raises:
        ValueError: the counts and the indices don't fit, or an index is out of range.

    Returns:
        str: handle of the stage.
    """
    if sum(face_vertex_counts) != len(face_vertex_indices):
        raise ValueError(
            f"The faces have {sum(face_vertex_counts)} corners, "
            f"there are {len(face_vertex_indices)} indices"
        )
    if face_vertex_indices and not 0 <= min(face_vertex_indices) <= max(
        face_vertex_indices
    ) < len(points):
        raise ValueError(f"An index is outside of the {len(points)} points")
    mesh = UsdGeom.Mesh.Define(get_stage(stage), prim_path)
    mesh.GetPointsAttr().Set(
        Vt.Vec3fArray([Gf.Vec3f(*_vector(p, "point")) for p in points])
    )
    mesh.GetFaceVertexCountsAttr().Set(Vt.IntArray([int(c) for c in face_vertex_counts]))
    mesh.GetFaceVertexIndicesAttr().Set(
        Vt.IntArray([int(i) for i in face_vertex_indices])
    )
    return stage

    def find_prims(stage: str, name_pattern: str, type_name: str = "") -> list[str]:
    """Find prims by name.

    Args:
        stage (str): stage handle.
        name_pattern (str): regular expression searched in the name of the prims, not
            their whole path.
        type_name (str): only the prims of this schema type, all if empty.

    Returns:
        list[str]: paths of the prims that match.
    """
    pattern = re.compile(name_pattern)
    return [
        str(prim.GetPath())
        for prim in get_stage(stage).Traverse()
        if pattern.search(prim.GetName())
        and (not type_name or prim.GetTypeName() == type_name)
    ]

    def set_visibility(stage: str, prim_path: str, visible: bool = True) -> str:
    """Show or hide a prim and what is below it.

    Args:
        stage (str): stage handle.
        prim_path (str): path of the prim.
        visible (bool): whether the prim is visible.

    Raises:
        ValueError: no prim at the path, or it has no visibility.

    Returns:
        str: handle of the stage.
    """
    imageable = UsdGeom.Imageable(get_prim(stage, prim_path))
    if not imageable:
        raise ValueError(f"{prim_path} has no visibility, it is not Imageable")
    if visible:
        imageable.MakeVisible()
    else:
        imageable.MakeInvisible()
    return stage
